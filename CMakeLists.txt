cmake_minimum_required(VERSION 3.22)
project(HALt VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(HALT_SANITIZERS "Enable ASan and UBSan on project code" OFF)
option(HALT_TIDY "Run clang-tidy during compilation" OFF)
find_package(Threads REQUIRED)
find_package(Protobuf REQUIRED)
find_package(gRPC CONFIG REQUIRED)
find_package(GTest CONFIG REQUIRED)

add_library(halt_options INTERFACE)
if(MSVC)
  target_compile_options(halt_options INTERFACE /W4 /WX)
else()
  target_compile_options(halt_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()
if(HALT_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "HALT_SANITIZERS requires GCC or Clang on Linux")
  endif()
  target_compile_options(halt_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(halt_options INTERFACE -fsanitize=address,undefined)
endif()
if(HALT_TIDY)
  find_program(CLANG_TIDY clang-tidy REQUIRED)
endif()

set(GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GENERATED_DIR}")
add_custom_command(
  OUTPUT "${GENERATED_DIR}/halt.pb.cc" "${GENERATED_DIR}/halt.pb.h"
         "${GENERATED_DIR}/halt.grpc.pb.cc" "${GENERATED_DIR}/halt.grpc.pb.h"
  COMMAND protobuf::protoc
  ARGS --proto_path=${CMAKE_CURRENT_SOURCE_DIR}/proto --cpp_out=${GENERATED_DIR}
       --grpc_out=${GENERATED_DIR} --plugin=protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin>
       ${CMAKE_CURRENT_SOURCE_DIR}/proto/halt.proto
  DEPENDS proto/halt.proto VERBATIM)
add_library(halt_proto "${GENERATED_DIR}/halt.pb.cc" "${GENERATED_DIR}/halt.grpc.pb.cc")
target_include_directories(halt_proto SYSTEM PUBLIC "${GENERATED_DIR}")
target_link_libraries(halt_proto PUBLIC protobuf::libprotobuf gRPC::grpc++)

add_library(halt_core cpp/src/types.cpp cpp/src/events.cpp cpp/src/device.cpp cpp/src/instruments.cpp)
target_include_directories(halt_core PUBLIC cpp/include)
target_link_libraries(halt_core PUBLIC Threads::Threads halt_options)
add_library(halt_service cpp/src/service.cpp)
target_link_libraries(halt_service PUBLIC halt_core halt_proto)
add_executable(halt_server cpp/src/main.cpp)
target_link_libraries(halt_server PRIVATE halt_service)
if(HALT_TIDY)
  set_target_properties(halt_core halt_service halt_server PROPERTIES CXX_CLANG_TIDY "${CLANG_TIDY}")
endif()
enable_testing()
add_executable(halt_unit_tests tests/cpp/core_test.cpp)
target_link_libraries(halt_unit_tests PRIVATE halt_core GTest::gtest_main)
include(GoogleTest)
gtest_discover_tests(halt_unit_tests)
